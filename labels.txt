import sqlite3
import numpy as np
import pandas as pd
import streamlit as st
import tensorflow as tf
from PIL import Image

# ---------------------------------------------------------
# 1. DATENBANK (SQLite) INITIALLISIERUNG
# ---------------------------------------------------------
DB_NAME = "fundbuero.db"


def init_db():
    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute(
        """
        CREATE TABLE IF NOT EXISTS fundstuecke (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            titel TEXT NOT NULL,
            kategorie TEXT NOT NULL,
            fundort TEXT NOT NULL,
            raum TEXT NOT NULL,
            datum DATE DEFAULT CURRENT_DATE,
            status TEXT NOT NULL,
            kontakt TEXT,
            bild_data BLOB
        )
    """
    )
    conn.commit()
    conn.close()


init_db()

# ---------------------------------------------------------
# 2. KI-MODELL & KLASSIFIKATION (Keras / MobileNetV2)
# ---------------------------------------------------------
# Mapping-Beispiel für 100 Klassen auf lesbare Kategorien
CATEGORIES = [
    "Elektronik",
    "Kleidung",
    "Schulmaterial",
    "Tasche/Rucksack",
    "Schlüssel",
    "Sonstiges",
]


@st.cache_resource
def load_keras_model():
    """Lädt das vortrainierte Keras Model (MobileNetV2 Basis + Dense(100))."""
    try:
        # Ersetze 'keras_model.h5' durch den genauen Pfad deines Modells
        model = tf.keras.models.load_model("keras_model.h5")
        return model
    except Exception as e:
        return None


model = load_keras_model()


def predict_category(image: Image.Image) -> str:
    """Skaliert das Bild auf 224x224x3 und führt eine Kategorisierung durch."""
    if model is None:
        return "Elektronik"  # Fallback, falls Keras-Datei nicht gefunden wurde

    # 1. Bild auf 224x224 Pixel skalieren
    img_resized = image.resize((224, 224))
    img_array = np.array(img_resized)

    # Bei Bedarf Alpha-Kanal entfernen (RGB sicherstellen)
    if img_array.shape[-1] == 4:
        img_array = img_array[..., :3]

    # Preprocessing für MobileNetV2 / Standardisierung [0, 1] oder [-1, 1]
    img_array = np.expand_dims(img_array, axis=0).astype(np.float32)
    img_array = tf.keras.applications.mobilenet_v2.preprocess_input(img_array)

    # Prediction ausführen
    predictions = model.predict(img_array)
    class_index = np.argmax(predictions[0])

    # Zuordnung des Index (0-99) zu einer Kategorie
    category_index = class_index % len(CATEGORIES)
    return CATEGORIES[category_index]


# ---------------------------------------------------------
# 3. BENUTZEROBERFLÄCHE (Streamlit UI/UX)
# ---------------------------------------------------------
st.set_page_config(
    page_title="Schul-Fundbüro", page_icon="🔍", layout="wide"
)

st.title("🔍 Schul-Fundbüro App")

# Navigation über Seitenleiste
menu = ["Dashboard / Suche", "Etwas gefunden", "Gegenstand-Details"]
choice = st.sidebar.selectbox("Navigation", menu)

# --- SEITE 1: DASHBOARD & SUCHE ---
if choice == "Dashboard / Suche":
    st.header("📋 Übersicht aller Fundstücke")

    # Filter-Optionen
    col1, col2 = st.columns(2)
    with col1:
        search_term = st.text_input("🔎 Suche nach Titel, Ort oder Raum")
    with col2:
        category_filter = st.selectbox(
            "🏷️ Kategorie-Filter", ["Alle"] + CATEGORIES
        )

    # Datenbank-Abfrage
    conn = sqlite3.connect(DB_NAME)
    query = "SELECT id, titel, kategorie, fundort, raum, datum, status FROM fundstuecke WHERE 1=1"
    params = []

    if category_filter != "Alle":
        query += " AND kategorie = ?"
        params.append(category_filter)

    if search_term:
        query += (
            " AND (titel LIKE ? OR fundort LIKE ? OR raum LIKE ?)"
        )
        params.extend(
            [f"%{search_term}%", f"%{search_term}%", f"%{search_term}%"]
        )

    df = pd.read_sql_query(query, conn, params=params)
    conn.close()

    if df.empty:
        st.info("Keine Fundstücke gefunden.")
    else:
        # Tabellarische Übersicht & Kartenansicht
        st.dataframe(df, use_container_width=True)

        st.subheader("Bildergalerie")
        conn = sqlite3.connect(DB_NAME)
        c = conn.cursor()
        c.execute(
            "SELECT id, titel, kategorie, status, bild_data FROM fundstuecke"
        )
        items = c.fetchall()
        conn.close()

        cols = st.columns(3)
        for idx, item in enumerate(items):
            item_id, titel, kat, status, bild_blob = item

            # Filter auf Galeriesicht anwenden
            if category_filter != "Alle" and kat != category_filter:
                continue
            if (
                search_term
                and search_term.lower() not in titel.lower()
            ):
                continue

            with cols[idx % 3]:
                st.write(f"**{titel}** ({kat})")
                st.caption(f"Status: {status}")
                if bild_blob:
                    import io

                    img = Image.open(io.BytesIO(bild_blob))
                    st.image(img, use_column_width=True)
                st.write("---")

# --- SEITE 2: FORMULAR "ETWAS GEFUNDEN" ---
elif choice == "Etwas gefunden":
    st.header("➕ Neun Fundgegenstand eintragen")

    uploaded_file = st.file_uploader(
        "📸 Foto des Gegenstands hochladen",
        type=["jpg", "jpeg", "png"],
    )

    suggested_category = "Sonstiges"
    image_bytes = None

    if uploaded_file is not None:
        image = Image.open(uploaded_file)
        st.image(image, caption="Hochgeladenes Bild", width=250)

        # KI-Vorhersage ausführen
        suggested_category = predict_category(image)
        st.success(f"🤖 KI-Vorschlag für Kategorie: **{suggested_category}**")

        # Bild zu Bytes für SQLite BLOB konvertieren
        import io

        img_byte_arr = io.BytesIO()
        image.save(img_byte_arr, format=image.format or "PNG")
        image_bytes = img_byte_arr.getvalue()

    with st.form("found_item_form"):
        titel = st.text_input("Titel / Kurzbeschreibung")
        cat_index = (
            CATEGORIES.index(suggested_category)
            if suggested_category in CATEGORIES
            else 0
        )
        kategorie = st.selectbox(
            "Kategorie Bestätigen/Anpassen", CATEGORIES, index=cat_index
        )
        fundort = st.text_input("Fundort (z.B. Pausenhof, Turnhalle)")
        raum = st.text_input("Raum / Genauer Ort (z.B. Raum 102)")
        kontakt = st.text_input(
            "Dein Name/Kontakt (Findende/r - optional)"
        )

        submitted = st.form_submit_button("Fundstück Speichern")

        if submitted:
            if not titel or not fundort:
                st.error(
                    "Bitte fülle mindestens den Titel und den Fundort aus."
                )
            else:
                conn = sqlite3.connect(DB_NAME)
                c = conn.cursor()
                c.execute(
                    """
                    INSERT INTO fundstuecke (titel, kategorie, fundort, raum, status, kontakt, bild_data)
                    VALUES (?, ?, ?, ?, ?, ?, ?)
                """,
                    (
                        titel,
                        kategorie,
                        fundort,
                        raum,
                        "Gefunden",
                        kontakt,
                        image_bytes,
                    ),
                )
                conn.commit()
                conn.close()
                st.success(
                    f"Das Fundstück '{titel}' wurde erfolgreich eingetragen!"
                )

# --- SEITE 3: DETAILANSICHT & ANSPRUCH ERHEBEN ---
elif choice == "Gegenstand-Details":
    st.header("🔍 Details & 'Das gehört mir!'")

    conn = sqlite3.connect(DB_NAME)
    c = conn.cursor()
    c.execute(
        "SELECT id, titel FROM fundstuecke WHERE status = 'Gefunden'"
    )
    available_items = c.fetchall()
    conn.close()

    if not available_items:
        st.info("Aktuell gibt es keine offenen Fundstücke.")
    else:
        item_options = {
            f"ID {item[0]}: {item[1]}": item[0] for item in available_items
        }
        selected_option = st.selectbox(
            "Wähle ein Fundstück aus:", list(item_options.keys())
        )
        selected_id = item_options[selected_option]

        # Details laden
        conn = sqlite3.connect(DB_NAME)
        c = conn.cursor()
        c.execute(
            "SELECT titel, kategorie, fundort, raum, datum, status, kontakt, bild_data FROM fundstuecke WHERE id = ?",
            (selected_id,),
        )
        item_data = c.fetchone()
        conn.close()

        if item_data:
            (
                titel,
                kategorie,
                fundort,
                raum,
                datum,
                status,
                kontakt,
                bild_data,
            ) = item_data

            col1, col2 = st.columns([1, 2])
            with col1:
                if bild_data:
                    import io

                    img = Image.open(io.BytesIO(bild_data))
                    st.image(img, use_column_width=True)
                else:
                    st.write("*(Kein Bild verfügbar)*")

            with col2:
                st.subheader(titel)
                st.write(f"**Kategorie:** {kategorie}")
                st.write(f"**Fundort:** {fundort} ({raum})")
                st.write(f"**Gefunden am:** {datum}")
                st.write(f"**Status:** {status}")

                st.write("---")
                if st.button("✋ Das gehört mir!"):
                    conn = sqlite3.connect(DB_NAME)
                    c = conn.cursor()
                    c.execute(
                        "UPDATE fundstuecke SET status = 'Abgeholt' WHERE id = ?",
                        (selected_id,),
                    )
                    conn.commit()
                    conn.close()
                    st.balloons()
                    st.success(
                        "Vielen Dank! Der Status wurde auf 'Abgeholt' aktualisiert. Bitte melde dich im Sekretariat zur Abholung."
                    )
